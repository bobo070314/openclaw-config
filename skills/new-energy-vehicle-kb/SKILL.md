### 续航计算
```python
def range_estimate(battery_kwh, consumption, temperature=25):
    """
    估算续航
    
    Parameters:
    - battery_kwh: 电池容量(度)
    - consumption: 电耗(kWh/100km)，CLTC/NEDC/WLTP
    - temperature: 环境温度°C
    """
    temp_factor = 1.0
    if temperature < 0:
        temp_factor = 0.55  # 零下损耗45%
    elif temperature < 10:
        temp_factor = 0.70  # 0-10°C损耗30%
    elif temperature < 20:
        temp_factor = 0.85  # 10-20°C损耗15%
    elif 20 <= temperature <= 30:
        temp_factor = 1.0
    elif temperature > 35:
        temp_factor = 0.90  # 高温开空调损耗10%
    
    # 实际可用电量（保留10%低电提醒）
    usable = battery_kwh * 0.90
    
    range_km = usable / consumption * 100
    
    # 高速因素（120km/h电耗增加30%）
    range_real = range_km * temp_factor
    
    return {
        "标称续航": f"{battery_kwh / consumption * 100:.0f}km",
        "实际续航": f"{range_real:.0f}km",
        "折扣率": f"{temp_factor*100:.0f}%",
        "温度影响": f"{'极差' if temp_factor < 0.6 else '差' if temp_factor < 0.75 else '一般' if temp_factor < 0.9 else '良好'}"
    }

# 例：80度电池，15kWh/100km，冬天5°C
print(range_estimate(80, 15, 5))
# 输出：标称533km → 实际304km（57折）
```

def calc_cost_per_100km(consumption=15, rate=0.3):
    """每百公里电费"""
    return round(consumption * rate, 2)

def calc_vs_gas(ev_kwh_per_100km=15, gas_l_per_100km=8, gas_price=8.0):
    """电车vs油车对比"""
    ev_cost = ev_kwh_per_100km * 0.3  # 家庭谷电
    gas_cost = gas_l_per_100km * gas_price
    return {
        "每百公里电费": f"{ev_cost:.0f}元",
        "每百公里油费": f"{gas_cost:.0f}元",
        "节省": f"{(gas_cost - ev_cost):.0f}元/百公里",
        "年省(2万km)": f"{(gas_cost - ev_cost) * 200:.0f}元"
    }

## 五、电池保养
```
最佳充电区间：20%-80%（三元锂），15%-100%（磷酸铁锂）
避免：长期满电（三元锂）、长期亏电（<15%）、快充太频繁
推荐：磷酸铁锂每周满充一次（用于均衡电池）
气温骤降时多充5%-10%（应对热管理消耗）
长期停放保持50%-60%电量
```
