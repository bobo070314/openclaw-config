```python
        "特征": ["精力充沛", "气色好", "不易生病"],
        "调理": ["保持现有生活习惯"],
        "食疗": "均衡饮食即可"
    }
}

def diagnose_constitution(symptoms):
    """根据症状判断体质类型"""
    scores = {k: 0 for k in CONSTITUTION_TYPES}
    for s in symptoms:
        for t, info in CONSTITUTION_TYPES.items():
            if s in info["特征"]:
                scores[t] += 1
    sorted_types = sorted(scores.items(), key=lambda x: -x[1])
    return [t for t, s in sorted_types if s > 0]
```
